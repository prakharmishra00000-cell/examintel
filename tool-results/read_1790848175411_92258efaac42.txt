   104→export function QuestionExplainer() {
   105→  const api = useApi();
   106→  const setContext = useAppStore((s) => s.setContext);
   107→  const saveItem = useAppStore((s) => s.saveItem);
   108→
   109→  const [question, setQuestion] = useState<string>("");
   110→  const [explanation, setExplanation] = useState<QuestionExplanation | null>(null);
   111→  const [loading, setLoading] = useState<boolean>(false);
   112→  const [error, setError] = useState<string | null>(null);
   113→
   114→  // progressive reveal state — Level 1 always shown, others gated
   115→  const [showL2, setShowL2] = useState(false);
   116→  const [showL3, setShowL3] = useState(false);
   117→  const [showL4, setShowL4] = useState(false);
   118→  const [showL5, setShowL5] = useState(false);
   119→
   120→  async function handleExplain(q?: string) {
   121→    const text = (q ?? question).trim();
   122→    if (!text) {
   123→      toast.error("Please paste a question first");
   124→      return;
   125→    }
   126→    setLoading(true);
   127→    setError(null);
   128→    // reset reveal state for new explanation
   129→    setShowL2(false);
   130→    setShowL3(false);
   131→    setShowL4(false);
   132→    setShowL5(false);
   133→    setExplanation(null);
   134→
   135→    const res = await api.call<{ explanation: QuestionExplanation; error?: string }>(
   136→      "/api/question/explain",
   137→      { question: text }
   138→    );
   139→
   140→    if (!res || !res.explanation || res.error) {
   141→      setError(res?.error ?? "AI did not return an explanation.");
   142→      setLoading(false);
   143→      return;
   144→    }
   145→    setExplanation(res.explanation);
   146→    setContext("Question: " + text.slice(0, 60), "explanation");
   147→    setLoading(false);
   148→  }
   149→
   150→  function handleExampleClick(q: string) {
   151→    setQuestion(q);
   152→    handleExplain(q);
   153→  }
   154→
   155→  function handleReset() {
   156→    setExplanation(null);
   157→    setError(null);
   158→    setQuestion("");
   159→    setShowL2(false);
   160→    setShowL3(false);
   161→    setShowL4(false);
   162→    setShowL5(false);
   163→  }
   164→
   165→  function handleSave() {
   166→    if (!explanation) return;
   167→    const title = (explanation.originalQuestion || question).slice(0, 60);
   168→    const summary = `${explanation.topic ?? "Unknown topic"} · ${explanation.difficulty ?? "—"}`;
   169→    saveItem({
   170→      type: "explanation",
   171→      title,
   172→      summary,
   173→      data: explanation,
   174→    });
   175→    toast.success("Saved to My Research", { description: summary });
   176→  }
   177→
   178→  return (
   179→    <div className="space-y-6">
   180→      {/* ---------- Header ---------- */}
   181→      <header className="space-y-1.5">
   182→        <div className="flex items-center gap-2.5">
   183→          <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white shadow-md">
   184→            <HelpCircle className="h-5 w-5" />
   185→          </div>
   186→          <h1 className="text-2xl font-bold tracking-tight">AI Question Explainer</h1>
   187→        </div>
   188→        <p className="text-muted-foreground text-sm max-w-3xl">
   189→          Understand any question progressively — <span className="text-amber-600 dark:text-amber-400 font-medium">Quick Hint</span> →{" "}
   190→          <span className="text-violet-600 dark:text-violet-400 font-medium">Concept</span> →{" "}
   191→          <span className="text-sky-600 dark:text-sky-400 font-medium">Detailed Solution</span> →{" "}
   192→          <span className="text-emerald-600 dark:text-emerald-400 font-medium">Exam Shortcut</span> →{" "}
   193→          <span className="text-fuchsia-600 dark:text-fuchsia-400 font-medium">Learning Insight</span>.
   194→        </p>
   195→      </header>
   196→
   197→      {/* ---------- Input form ---------- */}
   198→      <Card className="border-violet-500/20">
   199→        <CardContent className="p-5 space-y-4">
   200→          <Textarea
   201→            value={question}
   202→            onChange={(e) => setQuestion(e.target.value)}
   203→            placeholder="Paste your question here. e.g. A train 150m long passes a pole in 15 seconds. Find its speed."
   204→            className="min-h-[120px] resize-y text-sm"
   205→            aria-label="Question to explain"
   206→          />
   207→          <div className="flex flex-wrap items-center gap-3">
   208→            <Button
   209→              onClick={() => handleExplain()}
   210→              disabled={loading || !question.trim()}
   211→              className="gap-2 bg-gradient-to-r from-violet-500 to-fuchsia-500 text-white shadow-md hover:from-violet-600 hover:to-fuchsia-600 hover:shadow-lg transition-all"
   212→            >
   213→              <Sparkles className="h-4 w-4" />
   214→              {loading ? "Explaining..." : "Explain Question"}
   215→            </Button>
   216→            {explanation && (
   217→              <Button variant="outline" onClick={handleReset} className="gap-2">
   218→                <RotateCcw className="h-4 w-4" />
   219→                New Question
   220→              </Button>
   221→            )}
   222→            <span className="text-xs text-muted-foreground ml-auto">
   223→              {question.length} chars
   224→            </span>
   225→          </div>
   226→
   227→          {/* example chips */}
   228→          <div className="space-y-2">
   229→            <p className="text-xs font-medium text-muted-foreground">Try an example:</p>
   230→            <div className="flex flex-wrap gap-2">
   231→              {EXAMPLE_QUESTIONS.map((q, i) => (
   232→                <button
   233→                  key={i}
   234→                  onClick={() => handleExampleClick(q)}
   235→                  disabled={loading}
   236→                  className="rounded-full border border-border bg-background px-3 py-1 text-xs text-muted-foreground transition-all hover:border-violet-500/40 hover:text-foreground hover:bg-violet-500/5 disabled:opacity-50 disabled:cursor-not-allowed text-left"
   237→                  title={q}
   238→                >
   239→                  {q.length > 56 ? q.slice(0, 56) + "..." : q}
   240→                </button>
   241→              ))}
   242→            </div>
   243→          </div>
   244→        </CardContent>
   245→      </Card>
   246→
   247→      {/* ---------- Loading ---------- */}
   248→      {loading && (
   249→        <Card>
   250→          <CardContent className="p-6">
   251→            <LoadingState label="AI is preparing a 5-level progressive explanation..." />
   252→          </CardContent>
   253→        </Card>
   254→      )}
   255→
   256→      {/* ---------- Error ---------- */}
   257→      {error && !loading && (
   258→        <Card className="border-destructive/30 bg-destructive/5">
   259→          <CardContent className="p-5">
   260→            <div className="flex items-start gap-3">
   261→              <AlertTriangle className="h-5 w-5 text-destructive mt-0.5 shrink-0" />
   262→              <div className="flex-1">
   263→                <p className="font-medium text-destructive">Explanation failed</p>
   264→                <p className="mt-1 text-sm text-muted-foreground">{error}</p>
   265→                <Button variant="outline" size="sm" className="mt-3" onClick={() => handleExplain()}>
   266→                  Try again
   267→                </Button>
   268→              </div>
   269→            </div>
   270→          </CardContent>
   271→        </Card>
   272→      )}
   273→
   274→      {/* ---------- Empty state ---------- */}
   275→      {!explanation && !loading && !error && (
   276→        <EmptyState
   277→          icon={HelpCircle}
   278→          title="No explanation yet"
   279→          description="Paste any exam question above (or pick an example) and the AI will explain it across five progressive levels — from a quick hint to a learning insight."
   280→        />
   281→      )}
   282→
   283→      {/* ---------- Result ---------- */}
   284→      {explanation && !loading && (
   285→        <div className="space-y-5">
   286→          {/* Metadata */}
   287→          <MetadataBar explanation={explanation} />
   288→
   289→          {/* Original question echo */}
   290→          <Card className="bg-muted/30 border-dashed">
   291→            <CardContent className="p-4">
   292→              <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground mb-1.5 flex items-center gap-1.5">
   293→                <BookOpen className="h-3.5 w-3.5" /> Original Question
   294→              </p>
   295→              <p className="text-sm leading-relaxed">{explanation.originalQuestion}</p>
   296→            </CardContent>
   297→          </Card>
   298→
   299→          {/* Level 1 — Quick Hint (always visible) */}
   300→          <LevelCard
   301→            level={1}
   302→            title="Quick Hint"
   303→            icon={Lightbulb}
   304→            tagline="A small nudge — try it before revealing more."
   305→          >
   306→            <p className="text-sm leading-relaxed">{explanation.levels.level1_quickHint}</p>
   307→          </LevelCard>
   308→
   309→          {/* Level 2 — Concept */}
   310→          {showL2 && (
   311→            <LevelCard
   312→              level={2}
   313→              title="Concept"
   314→              icon={Brain}
   315→              tagline="Understand the underlying rule before solving."
   316→            >
   317→              <div className="space-y-4">
   318→                <div>
   319→                  <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground mb-1">Concept</p>
   320→                  <p className="text-sm font-medium">{explanation.levels.level2_concept.conceptName}</p>
   321→                </div>
   322→                <div className="grid gap-3 sm:grid-cols-2">
   323→                  <Field label="Core Rule" value={explanation.levels.level2_concept.coreRule} icon={Tag} accent={ACCENTS[2]} />
   324→                  <Field label="Why It Applies" value={explanation.levels.level2_concept.whyItApplies} icon={GitBranch} accent={ACCENTS[2]} />
   325→                  {explanation.levels.level2_concept.shortcut && (
   326→                    <Field label="Shortcut" value={explanation.levels.level2_concept.shortcut} icon={Zap} accent={ACCENTS[2]} />
   327→                  )}
   328→                  {explanation.levels.level2_concept.commonMistake && (
   329→                    <Field label="Common Mistake" value={explanation.levels.level2_concept.commonMistake} icon={AlertTriangle} accent={ACCENTS[2]} />
   330→                  )}
   331→                </div>
   332→              </div>
   333→            </LevelCard>
   334→          )}
   335→
   336→          {/* Level 3 — Detailed Solution */}
   337→          {showL3 && (
   338→            <LevelCard
   339→              level={3}
   340→              title="Detailed Solution"
   341→              icon={ListOrdered}
   342→              tagline="Step-by-step walkthrough with formula and calculation."
   343→            >
   344→              <div className="space-y-4">
   345→                {/* Steps */}
   346→                <ol className="space-y-2">
   347→                  {explanation.levels.level3_detailedSolution.steps.map((step, i) => (
   348→                    <li key={i} className="flex gap-3 text-sm">
   349→                      <span className={cn("flex h-5 w-5 shrink-0 items-center justify-center rounded-full bg-gradient-to-br text-[10px] font-bold text-white", ACCENTS[3].gradient)}>
   350→                        {i + 1}
   351→                      </span>
   352→                      <span className="leading-relaxed pt-0.5">{step}</span>
   353→                    </li>
   354→                  ))}
   355→                </ol>
   356→
   357→                {/* Formula */}
   358→                {explanation.levels.level3_detailedSolution.formula && (
   359→                  <div className="space-y-1">
   360→                    <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground flex items-center gap-1.5">
   361→                      <Calculator className="h-3.5 w-3.5" /> Formula
   362→                    </p>
   363→                    <pre className={cn("rounded-md border px-3 py-2 font-mono text-sm overflow-x-auto", ACCENTS[3].bg, ACCENTS[3].border)}>
   364→                      {explanation.levels.level3_detailedSolution.formula}
   365→                    </pre>
   366→                  </div>
   367→                )}
   368→
   369→                {/* Substitution + Calculation */}
   370→                {(explanation.levels.level3_detailedSolution.substitution || explanation.levels.level3_detailedSolution.calculation) && (
   371→                  <div className="grid gap-3 sm:grid-cols-2">
   372→                    {explanation.levels.level3_detailedSolution.substitution && (
   373→                      <div className="space-y-1">
   374→                        <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Substitution</p>
   375→                        <code className={cn("block rounded-md border px-3 py-2 font-mono text-sm", ACCENTS[3].bg, ACCENTS[3].border)}>
   376→                          {explanation.levels.level3_detailedSolution.substitution}
   377→                        </code>
   378→                      </div>
   379→                    )}
   380→                    {explanation.levels.level3_detailedSolution.calculation && (
   381→                      <div className="space-y-1">
   382→                        <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Calculation</p>
   383→                        <code className={cn("block rounded-md border px-3 py-2 font-mono text-sm", ACCENTS[3].bg, ACCENTS[3].border)}>
   384→                          {explanation.levels.level3_detailedSolution.calculation}
   385→                        </code>
   386→                      </div>
   387→                    )}
   388→                  </div>
   389→                )}
   390→
   391→                {/* Final Answer */}
   392→                <div className={cn("rounded-lg border px-4 py-3 flex items-center gap-3", ACCENTS[3].bg, ACCENTS[3].border)}>
   393→                  <CheckCircle2 className={cn("h-5 w-5 shrink-0", ACCENTS[3].text)} />
   394→                  <div className="flex-1">
   395→                    <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Final Answer</p>
   396→                    <p className={cn("text-base font-bold", ACCENTS[3].text)}>{explanation.levels.level3_detailedSolution.finalAnswer}</p>
   397→                  </div>
   398→                </div>
   399→
   400→                {/* Explanation */}
   401→                <div>
   402→                  <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground mb-1">Explanation</p>
   403→                  <p className="text-sm leading-relaxed text-muted-foreground">{explanation.levels.level3_detailedSolution.explanation}</p>
   404→                </div>
   405→              </div>
   406→            </LevelCard>
   407→          )}
   408→
   409→          {/* Level 4 — Exam Shortcut (optional) */}
   410→          {showL4 && explanation.levels.level4_examShortcut && (
   411→            <LevelCard
   412→              level={4}
   413→              title="Exam Shortcut"
   414→              icon={Zap}
   415→              tagline="Faster path — save precious seconds on exam day."
   416→            >
   417→              <div className="space-y-4">
   418→                <Field label="Method" value={explanation.levels.level4_examShortcut.method} icon={Zap} accent={ACCENTS[4]} />
   419→                <div className="grid gap-3 sm:grid-cols-2">
   420→                  {explanation.levels.level4_examShortcut.mentalCalculation && (
   421→                    <Field label="Mental Calculation" value={explanation.levels.level4_examShortcut.mentalCalculation} icon={Calculator} accent={ACCENTS[4]} />
   422→                  )}
   423→                  {explanation.levels.level4_examShortcut.eliminationTechnique && (
   424→                    <Field label="Elimination Technique" value={explanation.levels.level4_examShortcut.eliminationTechnique} icon={AlertTriangle} accent={ACCENTS[4]} />
   425→                  )}
   426→                </div>
   427→                <Field label="Time-Saving Approach" value={explanation.levels.level4_examShortcut.timeSavingApproach} icon={CheckCircle2} accent={ACCENTS[4]} />
   428→              </div>
   429→            </LevelCard>
   430→          )}
   431→
   432→          {/* Level 5 — Learning Insight */}
   433→          {showL5 && (
   434→            <LevelCard
   435→              level={5}
   436→              title="Learning Insight"
   437→              icon={GraduationCap}
   438→              tagline="What to remember going forward."
   439→            >
   440→              <p className="text-sm leading-relaxed">{explanation.levels.level5_learningInsight}</p>
   441→            </LevelCard>
   442→          )}
   443→
   444→          {/* Reveal buttons */}
   445→          <Card className="bg-muted/20">
   446→            <CardContent className="p-4">
   447→              <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground mb-2.5">Reveal next level</p>
   448→              <div className="flex flex-wrap gap-2">
   449→                <RevealButton label="Show Concept" icon={Brain} accent={ACCENTS[2]} shown={showL2} onClick={() => setShowL2(true)} disabled={showL2} />
   450→                <RevealButton label="Show Detailed Solution" icon={ListOrdered} accent={ACCENTS[3]} shown={showL3} onClick={() => setShowL3(true)} disabled={showL3} />
   451→                {explanation.levels.level4_examShortcut && (
   452→                  <RevealButton label="Show Exam Shortcut" icon={Zap} accent={ACCENTS[4]} shown={showL4} onClick={() => setShowL4(true)} disabled={showL4} />

   454→                <RevealButton label="Show Learning Insight" icon={GraduationCap} accent={ACCENTS[5]} shown={showL5} onClick={() => setShowL5(true)} disabled={showL5} />
   455→              </div>
   456→            </CardContent>
   457→          </Card>
   458→
   459→          {/* Sources */}
   460→          <div className="space-y-2">
   461→            <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Sources</p>
   462→            <SourceBadgeList sources={explanation.sources} />
   463→          </div>
   464→
   465→          {/* Save action */}
   466→          <div className="flex justify-end">
   467→            <Button onClick={handleSave} className="gap-2 bg-gradient-to-r from-violet-500 to-fuchsia-500 text-white shadow-md hover:from-violet-600 hover:to-fuchsia-600">
   468→              <Save className="h-4 w-4" />
   469→              Save to My Research
   470→            </Button>
   471→          </div>
   472→        </div>
   473→      )}
   474→    </div>
   475→  );
   476→}

   482→function MetadataBar({ explanation }: { explanation: QuestionExplanation }) {
   483→  return (
   484→    <Card className="border-border/60">
   485→      <CardContent className="p-4 space-y-3">
   486→        <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground flex items-center gap-1.5">

   493→            </Badge>
   494→          )}
   495→          {explanation.topic && (
   496→            <Badge variant="outline" className="gap-1 border-sky-500/30 text-sky-600 dark:text-sky-400 bg-sky-500/10">
   497→              <Tag className="h-3 w-3" /> {explanation.topic}
   498→            </Badge>
   499→          )}
   500→          {explanation.subtopic && (
   501→            <Badge variant="outline" className="gap-1 border-cyan-500/30 text-cyan-600 dark:text-cyan-400 bg-cyan-500/10">
   502→              <Layers className="h-3 w-3" /> {explanation.subtopic}

   528→                  {c}
   529→                </Badge>
   530→              ))}
   531→            </div>
   532→          </div>
   533→        )}
   534→
   535→        {explanation.prerequisites && explanation.prerequisites.length > 0 && (
   536→          <div className="space-y-1">
   537→            <p className="text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">Prerequisites</p>

   541→                  {c}
   542→                </Badge>
   543→              ))}
   544→            </div>
   545→          </div>
   546→        )}
   547→      </CardContent>
   548→    </Card>
   549→  );
   550→}
