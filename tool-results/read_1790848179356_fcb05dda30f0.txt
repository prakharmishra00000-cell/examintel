    19→  async json<T>(system: string, user: string, schemaHint?: string): Promise<T> {
    20→    // Try to return canned structured data based on the request keywords
    21→    const req = (system + " " + user).toLowerCase();
    22→    if (req.includes("exam research") || req.includes("exam intelligence report") || req.includes("researchexam")) {
    23→      return mockExamResearch(user) as unknown as T;
    24→    }
    25→    if (req.includes("comparison") || req.includes("compare")) {
    26→      return mockExamComparison(user) as unknown as T;
    27→    }
    28→    if (req.includes("dependency") || req.includes("prerequisite")) {
    29→      return mockDependencyMap(user) as unknown as T;
    30→    }
    31→    if (req.includes("question explainer") || req.includes("explain the question") || req.includes("questionexplanation")) {
    32→      return mockQuestionExplanation(user) as unknown as T;
    33→    }
    34→    if (req.includes("evolution") || req.includes("evolve")) {
    35→      return mockQuestionEvolution(user) as unknown as T;
    36→    }
    37→    if (req.includes("paper") || req.includes("question paper")) {
    38→      return mockPaper(user) as unknown as T;
    39→    }
    40→    if (req.includes("pdf") && req.includes("mcq")) {
    41→      return mockMCQ(user) as unknown as T;
    42→    }
    43→    if (req.includes("pdf") || req.includes("document")) {
    44→      return mockPdfAnalysis(user) as unknown as T;
    45→    }
    46→    if (req.includes("preparation") || req.includes("simulator")) {
    47→      return mockPreparation(user) as unknown as T;
    48→    }
    49→    if (req.includes("multi-exam") || req.includes("optimizer")) {
    50→      return mockMultiExam(user) as unknown as T;
    51→    }
    52→    if (req.includes("performance") || req.includes("analyse")) {
    53→      return mockPerformance(user) as unknown as T;
    54→    }
    55→    // generic fallback
    56→    return extractJson<T>(`{"note":"Mock provider active. Configure OPENAI_API_KEY on Vercel for real AI.","prompt":${JSON.stringify(user.slice(0,200))}}`);
    57→  }
    58→
    59→  async chat(messages: ChatCompletionMessage[]): Promise<string> {
    60→    const last = messages[messages.length - 1]?.content ?? "";
    61→    return `**Mock mode active.** I received your message: "${last.slice(0, 150)}".\n\nTo enable real AI responses, set \`OPENAI_API_KEY\` in your Vercel project's Environment Variables. See the "API Keys" guide in the footer.`;
    62→  }
