   311→function mockQuestionExplanation(query: string): unknown {
   312→  // Extract the actual question. The route wraps it in triple-quotes:
   313→  //   QUESTION:\n"""\n<actual question>\n"""
   314→  let question = "";
   315→  const triple = query.match(/"""\s*\n([\s\S]*?)\n\s*"""/);
   316→  if (triple && triple[1]) {
   317→    question = triple[1].trim().slice(0, 300);
   318→  } else {
   319→    // fallback: try single-quoted
   320→    const m = query.match(/question[:\s]*["']([^"']+)["']/i);
   321→    if (m && m[1]) question = m[1].trim().slice(0, 300);
   322→  }
   323→  if (!question || question.length < 5) question = "A train travels 120 km in 2 hours. What is its speed?";
   324→
   325→  // Detect question type for relevant canned explanation
   326→  const q = question.toLowerCase();
   327→  let profile: {
   328→    subject: string; topic: string; subtopic: string; concept: string; difficulty: "Easy"|"Medium"|"Hard";
   329→    related: string[]; prereq: string[]; hint: string; conceptName: string; coreRule: string; why: string;
   330→    shortcut: string; mistake: string; steps: string[]; formula: string; substitution: string; calc: string;
   331→    answer: string; explanation: string; method: string; mental: string; elim: string; timeSave: string; insight: string;
   332→  };
   333→
   334→  if (/(solve|find|value of|equation|=)/.test(q) && /[a-z]\s*[-+]\s*\d/.test(q)) {

   899→// ============================================================
   900→// Formula Sheet lookup — searches the built-in SEED_FORMULAS by
   901→// keyword (name / topic / subject match). Returns 3–5 matches, or
   902→// all formulas when no keyword in the query matches anything.
   903→// ============================================================
   904→function mockFormulasLookup(query: string): unknown {
   905→  const lower = query.toLowerCase();
   906→
   907→  // Tokenize the query into candidate keywords (drop short / stop words).
   908→  const stop = new Set([
   909→    "the", "a", "an", "of", "for", "and", "or", "to", "in", "is", "are",
   910→    "what", "whats", "formula", "formulas", "formulae", "lookup", "search",
   911→    "find", "me", "please", "give", "show", "list", "all", "with",
   912→  ]);
   913→  const tokens = lower
   914→    .split(/[^a-z0-9&+-]+/i)
   915→    .map((t) => t.trim())
   916→    .filter((t) => t.length > 1 && !stop.has(t));
   917→
   918→  const matched = SEED_FORMULAS.filter((f) => {
   919→    const hay = `${f.subject} ${f.topic} ${f.name} ${f.description}`.toLowerCase();
   920→    return tokens.some((tok) => hay.includes(tok));
   921→  });
   922→
   923→  // De-dup by id (in case the same formula matches multiple tokens).
   924→  const seen = new Set<string>();
   925→  const unique = matched.filter((f) =>
   926→    seen.has(f.id) ? false : (seen.add(f.id), true)
   927→  );
   928→
   929→  // 3–5 matches when possible; if fewer keywords match, fall back to all.
   930→  let result = unique;
   931→  if (unique.length < 3) {
   932→    result = SEED_FORMULAS.slice();
   933→  }
   934→  const capped = result.slice(0, 5);
   935→
   936→  return {
   937→    formulas: capped.map((f) => ({
   938→      id: f.id,
   939→      subject: f.subject,
   940→      topic: f.topic,
   941→      name: f.name,
   942→      formula: f.formula,
   943→      description: f.description,
   944→      example: f.example,
   945→      difficulty: f.difficulty,
   946→    })),
   947→  };
   948→}
