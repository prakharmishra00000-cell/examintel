    67→function fmt(sec: number): string {
    68→  const s = Math.max(0, Math.floor(sec));
    69→  const m = Math.floor(s / 60);
    70→  const r = s % 60;
    71→  return `${String(m).padStart(2, "0")}:${String(r).padStart(2, "0")}`;
    72→}

    74→/** Local YYYY-MM-DD key for a date. */
    75→function dayKey(d: Date): string {
    76→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(
    77→    d.getDate()
    78→  ).padStart(2, "0")}`;
    79→}

    81→/** Get the Monday of the week containing `d` (local). */
    82→function startOfWeek(d: Date): Date {
    83→  const x = new Date(d);
    84→  const day = (x.getDay() + 6) % 7; // Mon = 0 ... Sun = 6
    85→  x.setHours(0, 0, 0, 0);
    86→  x.setDate(x.getDate() - day);
    87→  return x;
    88→}

    92→// ----- Sound (Web Audio) -------------------------------------------------

    94→let audioCtx: AudioContext | null = null;
    95→function getAudioCtx(): AudioContext | null {
    96→  if (typeof window === "undefined") return null;
    97→  if (!audioCtx) {
    98→    try {
    99→      const Ctor =
   100→        (window as any).AudioContext || (window as any).webkitAudioContext;
   101→      audioCtx = Ctor ? new Ctor() : null;
   102→    } catch {
   103→      audioCtx = null;
   104→    }
   105→  }
   106→  return audioCtx;
   107→}
   108→
   109→function playBeep(freq = 880, durationMs = 320) {
   110→  const ctx = getAudioCtx();
   111→  if (!ctx) return;
   112→  try {
   113→    if (ctx.state === "suspended") void ctx.resume();
   114→    const now = ctx.currentTime;
   115→    const osc = ctx.createOscillator();
   116→    const gain = ctx.createGain();
   117→    osc.type = "sine";
   118→    osc.frequency.value = freq;
   119→    gain.gain.setValueAtTime(0.0001, now);
   120→    gain.gain.exponentialRampToValueAtTime(0.18, now + 0.02);
   121→    gain.gain.exponentialRampToValueAtTime(0.0001, now + durationMs / 1000);
   122→    osc.connect(gain).connect(ctx.destination);
   123→    osc.start(now);
   124→    osc.stop(now + durationMs / 1000);
   125→  } catch {
   126→    /* ignore */
   127→  }
   128→}

   130→function playChime() {
   131→  playBeep(880, 240);
   132→  setTimeout(() => playBeep(1175, 320), 220);
   133→}

   135→// ----- Stats -------------------------------------------------------------
   136→function computeStreaks(sessions: StudySession[]) {
   137→  const completed = sessions.filter((s) => s.completed);
   138→  const daySet = new Set(completed.map((s) => dayKey(new Date(s.date))));
   139→
   140→  let current = 0;
   141→  // Walk backwards from today; allow today to be empty (streak still alive)
   142→  const cursor = new Date();
   143→  cursor.setHours(0, 0, 0, 0);
   144→  // If today has no session, start checking from yesterday for "current"
   145→  let probe = new Date(cursor);
   146→  if (!daySet.has(dayKey(probe))) {
   147→    probe.setDate(probe.getDate() - 1);
   148→  }
   149→  while (daySet.has(dayKey(probe))) {
   150→    current += 1;
   151→    probe.setDate(probe.getDate() - 1);
   152→  }
   153→
   154→  // Longest streak: walk all distinct sorted days
   155→  const sortedDays = Array.from(daySet).sort();
   156→  let longest = 0;
   157→  let run = 0;
   158→  let prev: Date | null = null;
   159→  for (const k of sortedDays) {
   160→    const d = new Date(k + "T00:00:00");
   161→    if (prev) {
   162→      const diffDays = Math.round(
   163→        (d.getTime() - prev.getTime()) / 86400000
   164→      );
   165→      if (diffDays === 1) run += 1;
   166→      else run = 1;
   167→    } else {
   168→      run = 1;
   169→    }
   170→    longest = Math.max(longest, run);
   171→    prev = d;
   172→  }
   173→
   174→  return { current, longest, totalSessions: completed.length };
   175→}
   176→
   177→function computeWeek(sessions: StudySession[]) {
   178→  const weekStart = startOfWeek(new Date());
   179→  const totals = [0, 0, 0, 0, 0, 0, 0];
   180→  let weekMinutes = 0;
   181→  for (const s of sessions) {
   182→    if (!s.completed) continue;
   183→    const d = new Date(s.date);
   184→    const diffDays = Math.floor(
   185→      (startOfWeek(d).getTime() - weekStart.getTime()) / 86400000
   186→    );
   187→    if (diffDays === 0) {
   188→      const idx = (d.getDay() + 6) % 7;
   189→      totals[idx] += s.durationMinutes;
   190→      weekMinutes += s.durationMinutes;
   191→    }
   192→  }
   193→  return { totals, weekMinutes };
   194→}

   209→  const [phase, setPhase] = useState<Phase>("work");
   210→  const [secondsLeft, setSecondsLeft] = useState(POMODORO_WORK);
   211→  const [running, setRunning] = useState(false);
   212→  // free-timer custom minutes
   213→  const [freeMinutes, setFreeMinutes] = useState(45);
   214→  const [freeInput, setFreeInput] = useState("45");
   215→
   216→  // confetti burst
   217→  const [burst, setBurst] = useState(false);
   218→  const burstTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   219→
   220→  // interval
   221→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   222→
   223→  // ---- total seconds for current timer (for progress ring) ----
   224→  const totalSeconds = useMemo(() => {
   225→    if (mode === "pomodoro") return phase === "work" ? POMODORO_WORK : POMODORO_BREAK;
   226→    return Math.max(1, Math.min(180, freeMinutes)) * 60;
   227→  }, [mode, phase, freeMinutes]);
   228→
   229→  // NOTE: secondsLeft is reset in event handlers (onModeChange, handleSkip,
   230→  // handleReset, applyFreeMinutes, handleComplete) — no useEffect needed.
   231→
   232→  // ---- cleanup on unmount ----
   233→  useEffect(() => {
   234→    return () => {
   235→      if (intervalRef.current) clearInterval(intervalRef.current);
   236→      if (burstTimer.current) clearTimeout(burstTimer.current);
   237→    };
   238→  }, []);
   239→
   240→  // ---- save a work session ----
   241→  const saveWorkSession = useCallback(
   242→    (durationMinutes: number, m: Mode, subj: string, top: string) => {
   243→      addSession({
   244→        subject: subj.trim() || "Untitled",
   245→        topic: top.trim() || "General",
   246→        durationMinutes,
   247→        date: new Date().toISOString(),
   248→        mode: m,
   249→        completed: true,
   250→      });
   251→    },
   252→    [addSession]
   253→  );
   254→
   255→  // ---- trigger completion: sound + toast + burst ----
   256→  const celebrate = useCallback((msg: string) => {
   257→    playChime();
   258→    setBurst(true);
   259→    if (burstTimer.current) clearTimeout(burstTimer.current);
   260→    burstTimer.current = setTimeout(() => setBurst(false), 1800);
   261→    toast.success(msg, {
   262→      description: "Great focus — session saved to your history.",
   263→      duration: 4000,
   264→    });
   265→  }, []);
   266→
   267→  // ---- handle phase completion ----
   268→  const handleComplete = useCallback(() => {
   269→    if (mode === "pomodoro") {
   270→      if (phase === "work") {
   271→        // Save the work session
   272→        saveWorkSession(25, "pomodoro", subject, topic);
   273→        celebrate("Pomodoro complete! Time for a 5-min break.");
   274→        setPhase("break");
   275→        setSecondsLeft(POMODORO_BREAK);
   276→        // keep running so break auto-starts
   277→      } else {
   278→        // Break finished — back to work, auto pause
   279→        playBeep(660, 200);
   280→        toast.info("Break over. Ready for the next pomodoro?", {
   281→          duration: 4000,
   282→        });
   283→        setPhase("work");
   284→        setSecondsLeft(POMODORO_WORK);
   285→        setRunning(false);
   286→      }
   287→    } else {
   288→      // Free timer complete
   289→      const mins = Math.max(1, Math.min(180, freeMinutes));
   290→      saveWorkSession(mins, "free", subject, topic);
   291→      celebrate(`Focus session complete — ${mins} min logged!`);
   292→      setRunning(false);
   293→      setSecondsLeft(mins * 60);
   294→    }
   295→  }, [mode, phase, subject, topic, freeMinutes, saveWorkSession, celebrate]);
   296→
   297→  // ---- interval effect ----
   298→  useEffect(() => {
   299→    if (!running) {
   300→      if (intervalRef.current) {
   301→        clearInterval(intervalRef.current);
   302→        intervalRef.current = null;
   303→      }
   304→      return;
   305→    }
   306→    intervalRef.current = setInterval(() => {
   307→      setSecondsLeft((prev) => {
   308→        if (prev <= 1) {
   309→          // we'll handle completion in the next tick to avoid setState-in-callback warning
   310→          return 0;
   311→        }
   312→        return prev - 1;
   313→      });
   314→    }, 1000);
   315→    return () => {
   316→      if (intervalRef.current) clearInterval(intervalRef.current);
   317→      intervalRef.current = null;
   318→    };
   319→  }, [running]);
   320→
   321→  // watch for hitting zero while running
   322→  useEffect(() => {
   323→    if (running && secondsLeft === 0) {
   324→      // Defer completion to escape the effect body (avoids
   325→      // cascading renders & setState-in-effect lint rule).
   326→      const id = setTimeout(() => handleComplete(), 0);
   327→      return () => clearTimeout(id);
   328→    }
   329→  }, [secondsLeft, running, handleComplete]);
   330→
   331→  // ---- controls ----
   332→  const canStart =
   333→    subject.trim().length > 0 && (mode === "pomodoro" || freeMinutes >= 1);

   337→      toast.error("Add a subject first", {
   338→        description: "Tag every focus session with a subject + topic.",
   339→      });
   340→      return;
   341→    }
   342→    // resume audio context on user gesture
   343→    const ctx = getAudioCtx();
   344→    if (ctx && ctx.state === "suspended") void ctx.resume();
   345→    setRunning((r) => !r);
   346→  };
   347→
   348→  const handleReset = () => {
   349→    setRunning(false);
   350→    if (mode === "pomodoro") {
   351→      setPhase("work");
   352→      setSecondsLeft(POMODORO_WORK);
   353→    } else {
   354→      const m = Math.max(1, Math.min(180, freeMinutes));
   355→      setSecondsLeft(m * 60);
   356→    }
   357→  };
   358→
   359→  const handleSkip = () => {
   360→    if (mode === "pomodoro") {
   361→      if (phase === "work") {
   362→        // Skipping work does NOT save (no completion)
   363→        setPhase("break");
   364→        setSecondsLeft(POMODORO_BREAK);
   365→      } else {
   366→        setPhase("work");
   367→        setSecondsLeft(POMODORO_WORK);
   368→        setRunning(false);
   369→      }
   370→    } else {
   371→      setRunning(false);
   372→      const m = Math.max(1, Math.min(180, freeMinutes));
   373→      setSecondsLeft(m * 60);
   374→    }
   375→  };
   376→
   377→  const onModeChange = (m: string) => {
   378→    setRunning(false);
   379→    const next = m as Mode;
   380→    setMode(next);
   381→    if (next === "pomodoro") {
   382→      setPhase("work");
   383→      setSecondsLeft(POMODORO_WORK);
   384→    } else {
   385→      const mn = Math.max(1, Math.min(180, freeMinutes));
   386→      setSecondsLeft(mn * 60);

   389→

   392→    if (isNaN(n) || n < 1) {

   397→    setFreeMinutes(clamped);
   398→    setFreeInput(String(clamped));
   399→    if (!running) setSecondsLeft(clamped * 60);

   407→  const todayKey = dayKey(new Date());
   408→  const todaySessions = useMemo(
   409→    () => sessions.filter((s) => dayKey(new Date(s.date)) === todayKey),
   410→    [sessions, todayKey]
   411→  );
   412→  const recentSessions = useMemo(() => sessions.slice(0, 20), [sessions]);
   413→
   414→  // ---- progress for ring ----
   415→  const progress = totalSeconds > 0 ? 1 - secondsLeft / totalSeconds : 0;
   416→  const radius = 110;
   417→  const circumference = 2 * Math.PI * radius;
   418→  const dashOffset = circumference * (1 - progress);
   419→
   420→  const ringGradient =
   421→    phase === "work"
   422→      ? "url(#gradViolet)"
   423→      : "url(#gradEmerald)";
   424→
   425→  return (
   426→    <div className="space-y-6">
   427→      {/* Header */}
   428→      <motion.div
   429→        initial={{ opacity: 0, y: 8 }}
   430→        animate={{ opacity: 1, y: 0 }}
   431→        transition={{ duration: 0.3 }}
   432→      >
   433→        <div className="flex items-center gap-3">
   434→          <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white shadow-lg shadow-violet-500/30">
   435→            <Timer className="h-5 w-5" />

   439→              Study Timer

   441→            <p className="text-sm text-muted-foreground">
   442→              Focus sessions with Pomodoro. Track streaks. Build consistency.
   443→            </p>
   444→          </div>
   445→        </div>
   446→      </motion.div>
   447→
   448→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   449→        {/* ---------------- LEFT: timer ---------------- */}
   450→        <Card className="lg:col-span-2 relative overflow-hidden">
