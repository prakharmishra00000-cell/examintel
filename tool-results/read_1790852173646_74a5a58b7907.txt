   276→export function McqGenerator() {
   277→  const api = useApi();
   278→  const setContext = useAppStore((s) => s.setContext);
   279→  const saveItem = useAppStore((s) => s.saveItem);
   280→
   281→  // step state

   283→
   284→  // form state
   285→  const [source, setSource] = useState<SourceKey>("Uploaded PDF");
   286→  const [content, setContent] = useState<string>("");
   287→  const [topic, setTopic] = useState<string>("");
   288→  const [customTopic, setCustomTopic] = useState<string>("");
   289→  const [difficulty, setDifficulty] = useState<DifficultyKey>("Mixed Difficulty");
   290→  const [questionCount, setQuestionCount] = useState<number>(5);
   291→  const [customCount, setCustomCount] = useState<string>("");
   292→  const [useCustomCount, setUseCustomCount] = useState<boolean>(false);
   293→  const [questionType, setQuestionType] = useState<QuestionTypeKey>("MCQ");
   294→  const [examName, setExamName] = useState<string>("");
   295→
   296→  // generation state
   297→  const [loading, setLoading] = useState<boolean>(false);
   298→  const [mcqSet, setMcqSet] = useState<MCQSet | null>(null);
   299→
   300→  // test-mode state
   301→  const [answers, setAnswers] = useState<Record<string, string>>({});
   302→  const [activeQ, setActiveQ] = useState<number>(0);
   303→  const [submitted, setSubmitted] = useState<boolean>(false);
   304→  const [elapsed, setElapsed] = useState<number>(0);
   305→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   306→
   307→  // derived mode
   308→  const mode = useMemo(() => deriveMode(content), [content]);
   309→
   310→  // effective question count
   311→  const effectiveCount = useCustomCount
   312→    ? Math.min(50, Math.max(1, parseInt(customCount, 10) || 0))
   313→    : questionCount;
   314→
   315→  // effective topic label (used for context/save)
   316→  const effectiveTopic = useMemo(() => {
   317→    if (topic && topic.trim()) return topic.trim();
   318→    if (customTopic && customTopic.trim()) return customTopic.trim();
   319→    if (source === "Uploaded PDF") return "Uploaded PDF";
   320→    if (source === "Syllabus") return "Syllabus";
   321→    return "Custom";
   322→  }, [topic, customTopic, source]);
   323→
   324→  // ----------------- timer -----------------
   325→  useEffect(() => {
   326→    if (mcqSet && !submitted) {
   327→      timerRef.current = setInterval(() => setElapsed((e) => e + 1), 1000);
   328→      return () => {
   329→        if (timerRef.current) clearInterval(timerRef.current);
   330→      };
   331→    }
   332→    if (submitted && timerRef.current) {
   333→      clearInterval(timerRef.current);
   334→      timerRef.current = null;
   335→    }
   336→  }, [mcqSet, submitted]);
   337→
   338→  const resetTestState = useCallback(() => {
   339→    setAnswers({});
   340→    setActiveQ(0);
   341→    setSubmitted(false);
   342→    setElapsed(0);
   343→    if (timerRef.current) {
   344→      clearInterval(timerRef.current);
   345→      timerRef.current = null;
   346→    }
   347→  }, []);
   348→
   349→  // ----------------- handlers -----------------
   350→  function handleNext() {
   351→    setStep((s) => Math.min(5, s + 1));
   352→  }
   353→
   354→  function handleBack(to?: number) {
   355→    setStep((s) => (typeof to === "number" ? to : Math.max(1, s - 1)));
   356→  }
   357→
   358→  function canProceedFromStep(s: number): boolean {
   359→    if (s === 1) {
   360→      if (source === "Uploaded PDF" || source === "Syllabus" || source === "Custom") {
   361→        return content.trim().length > 0;
   362→      }
   363→      if (source === "Topic") {
   364→        return topic.trim().length > 0 || customTopic.trim().length > 0;
   365→      }
   366→      return false;
   367→    }
   368→    return true;
   369→  }
   370→
   371→  function selectAnswer(qid: string, option: string) {
   372→    if (submitted) return;
   373→    setAnswers((prev) => ({ ...prev, [qid]: option }));
   374→  }
   375→
   376→  async function handleGenerate() {
   377→    setLoading(true);
   378→    resetTestState();
   379→    const body = {
   380→      source,
   381→      content,
   382→      topic: effectiveTopic,
   383→      difficulty: difficulty === "Mixed Difficulty" ? "Mixed" : difficulty,
   384→      questionCount: effectiveCount,

   386→      mode,

   404→  function handleSave() {

   408→    saveItem({
   409→      type: "mcq",
   410→      title: `${effectiveTopic} · ${mcqSet.questionCount}Q`,
   411→      summary: `${accuracy}% accuracy`,
   412→      data: { set: mcqSet, answers, elapsed, submitted },
   413→    });
   414→    toast.success("Saved to My Research", {
   415→      description: `${effectiveTopic} · ${mcqSet.questionCount}Q · ${accuracy}% accuracy`,
   416→    });
   417→  }

   422→    setStep(1);
   423→    setSource("Uploaded PDF");
   424→    setContent("");
   425→    setTopic("");
   426→    setCustomTopic("");

   435→  function handleUploadTxt(e: React.ChangeEvent<HTMLInputElement>) {
   436→    const file = e.target.files?.[0];
   437→    if (!file) return;
   438→    const reader = new FileReader();
   439→    reader.onload = () => {
   440→      const text = String(reader.result ?? "");
   441→      setContent(text);
   442→      toast.success(`Loaded ${file.name}`, { description: `${text.length.toLocaleString()} characters` });
   443→    };
   444→    reader.readAsText(file);
   445→  }
   446→
   447→  // ----------------- scoring -----------------
   448→  function computeScore() {
   449→    if (!mcqSet) return { correct: 0, incorrect: 0, unattempted: 0, total: 0, accuracy: 0 };
   450→    let correct = 0;
   451→    let incorrect = 0;
   452→    let unattempted = 0;
   453→    for (const m of mcqSet.mcqs) {
   454→      const user = answers[m.id];
   455→      if (!user) {
   456→        unattempted += 1;
   457→        continue;
   458→      }
   459→      // For "Multiple correct" correctAnswer is comma-separated; compare as sets
   460→      const correctSet = new Set(m.correctAnswer.split(",").map((s) => s.trim()).filter(Boolean));
   461→      const userSet = new Set(user.split(",").map((s) => s.trim()).filter(Boolean));
   462→      const same =
   463→        correctSet.size === userSet.size &&
   464→        [...correctSet].every((s) => userSet.has(s));
   465→      if (same) correct += 1;
   466→      else incorrect += 1;
   467→    }
   468→    const total = mcqSet.mcqs.length;
   469→    const attempted = total - unattempted;
   470→    const accuracy = attempted > 0 ? Math.round((correct / total) * 100) : 0;
   471→    return { correct, incorrect, unattempted, total, attempted, accuracy };
   472→  }
   473→
   474→  function answerStateFor(m: GeneratedMCQ): AnswerState {
   475→    const user = answers[m.id];
   476→    if (!user) return "unattempted";
   477→    const correctSet = new Set(m.correctAnswer.split(",").map((s) => s.trim()).filter(Boolean));
   478→    const userSet = new Set(user.split(",").map((s) => s.trim()).filter(Boolean));
   479→    const same =
   480→      correctSet.size === userSet.size &&
   481→      [...correctSet].every((s) => userSet.has(s));
   482→    return same ? "correct" : "wrong";
   483→  }
