    19→  async json<T>(system: string, user: string, schemaHint?: string): Promise<T> {
    20→    // Route primarily on the SCHEMA HINT (most reliable signal of expected shape),
    21→    // falling back to combined keyword search. This avoids mis-routing when a system
    22→    // prompt for one feature mentions words like "prerequisite" or "dependency".
    23→    const schema = (schemaHint ?? "").toLowerCase();
    24→    const req = (system + " " + user).toLowerCase();
    25→
    26→    if (schema.includes("examresearchreport") || req.includes("exam intelligence researcher") || req.includes("examresearch")) {
    27→      return mockExamResearch(user) as unknown as T;
    28→    }
    29→    if (schema.includes("examcomparisonreport") || req.includes("exam comparison engine") || req.includes("examcomparison")) {
    30→      return mockExamComparison(user) as unknown as T;
    31→    }
    32→    if (schema.includes("dependencymapreport") || req.includes("dependency mapper") || req.includes("prerequisite-dependency mapper")) {
    33→      return mockDependencyMap(user) as unknown as T;
    34→    }
    35→    if (schema.includes("questionexplanation") || req.includes("progressive question explainer") || req.includes("questionexplanation")) {
    36→      return mockQuestionExplanation(user) as unknown as T;
    37→    }
    38→    if (schema.includes("questionevolutionreport") || req.includes("question evolution engine") || req.includes("questionevolution")) {
    39→      return mockQuestionEvolution(user) as unknown as T;
    40→    }
    41→    if (schema.includes("generatedpaper") || req.includes("personalized question paper generator")) {
    42→      return mockPaper(user) as unknown as T;
    43→    }
    44→    if (schema.includes("mcqset") || req.includes("grounded mcq generator")) {
    45→      return mockMCQ(user) as unknown as T;
    46→    }
    47→    if (schema.includes("pdfanalysisreport") || req.includes("pdf exam analyzer")) {
    48→      return mockPdfAnalysis(user) as unknown as T;
    49→    }
    50→    if (schema.includes("preparationplan") || req.includes("adaptive preparation simulator")) {
    51→      return mockPreparation(user) as unknown as T;
    52→    }
    53→    if (schema.includes("multiexamplan") || req.includes("multi-exam preparation optimizer")) {

    55→    }
    56→    if (schema.includes("performanceanalysis") || req.includes("performance analysis")) {
