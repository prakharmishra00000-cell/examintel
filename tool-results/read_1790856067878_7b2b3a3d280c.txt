   187→// ============================================================
   188→// Aggregation core
   189→// ============================================================
   190→
   191→function aggregateTopics(
   192→  saved: SavedItem[],
   193→  sets: FlashcardSet[],
   194→  sessions: StudySession[],
   195→  entries: JournalEntry[],
   196→): Map<string, TopicAgg> {
   197→  const map = new Map<string, TopicAgg>();
   198→
   199→  function get(topicRaw: string): TopicAgg | null {
   200→    const topic = (topicRaw ?? "").trim();
   201→    if (!topic) return null;
   202→    let agg = map.get(topic);
   203→    if (!agg) {
   204→      agg = {
   205→        topic,
   206→        subject: null,
   207→        flashcards: [],
   208→        sessions: [],
   209→        journal: [],
   210→        questions: [],
   211→        fcScore: null,
   212→        qaScore: null,
   213→        activityBoost: 0,
   214→        blend: 0,
   215→        score: 0,
   216→        status: "Not Started",
   217→      };
   218→      map.set(topic, agg);
   219→    }
   220→    return agg;
   221→  }
   222→
   223→  // 1) Saved items
   224→  for (const item of saved) {
   225→    const data = (item.data ?? null) as Record<string, unknown> | null;
   226→    if (!data) continue;
   227→
   228→    if (item.type === "explanation") {
   229→      const t = str((data as { topic?: unknown }).topic);
   230→      if (!t) continue;
   231→      const agg = get(t);
   232→      if (!agg) continue;
   233→      const subj = str((data as { subject?: unknown }).subject);
   234→      if (subj && !agg.subject) agg.subject = subj;
   235→      const diff = str((data as { difficulty?: unknown }).difficulty);
   236→      agg.questions.push({
   237→        source: "explanation",
   238→        itemId: item.id,
   239→        title: item.title,
   240→        difficulty: diff,
   241→        hasAccuracy: false,
   242→      });
   243→    } else if (item.type === "evolution") {
   244→      const t = str((data as { topic?: unknown }).topic);
   245→      if (!t) continue;
   246→      const agg = get(t);
   247→      if (!agg) continue;
   248→      const subj = str((data as { subject?: unknown }).subject);
   249→      if (subj && !agg.subject) agg.subject = subj;
   250→      const diff = str((data as { difficulty?: unknown }).difficulty);
   251→      agg.questions.push({
   252→        source: "evolution",
   253→        itemId: item.id,
   254→        title: item.title,
   255→        difficulty: diff,
   256→        hasAccuracy: false,
   257→      });
   258→    } else if (item.type === "paper") {
   259→      // data shape: { paper: GeneratedPaper, analysis?: PerformanceAnalysis, answers? }
   260→      const paper = (data as { paper?: Record<string, unknown> }).paper;
   261→      const questions = Array.isArray(paper?.questions)
   262→        ? (paper!.questions as unknown as Array<Record<string, unknown>>)
   263→        : Array.isArray((data as { questions?: unknown }).questions)
   264→          ? ((data as { questions?: Array<Record<string, unknown>> }).questions as Array<Record<string, unknown>>)
   265→          : [];
   266→      const analysis = (data as { analysis?: PerformanceAnalysis }).analysis;
   267→      const topicWise = analysis?.topicWise ?? [];
   268→      for (const q of questions) {
   269→        const t = str(q?.topic);
   270→        if (!t) continue;
   271→        const agg = get(t);
   272→        if (!agg) continue;
   273→        const tw = topicWise.find(
   274→          (x) => x && x.topic && x.topic.trim().toLowerCase() === t.toLowerCase(),
   275→        );
   276→        const subj = str(q?.subject) ?? str((data as { subject?: unknown }).subject);
   277→        if (subj && !agg.subject) agg.subject = subj;
   278→        const diff = str(q?.difficulty);
   279→        const hasAcc = !!tw && typeof tw.correct === "number" && typeof tw.total === "number";
   280→        agg.questions.push({
   281→          source: "paper",
   282→          itemId: item.id,
   283→          title: item.title,
   284→          difficulty: diff,
   285→          correct: tw?.correct,
   286→          total: tw?.total,
   287→          hasAccuracy: hasAcc,
   288→        });
   289→      }
   290→    } else if (item.type === "mcq") {
   291→      // data shape: { set: { mcqs: [...] }, answers: Record<id, picked>, ... }
   292→      // OR legacy: { mcqs: [...] }
   293→      const setData = (data as { set?: Record<string, unknown> }).set;
   294→      const mcqs: Array<Record<string, unknown>> = Array.isArray(setData?.mcqs)
   295→        ? (setData!.mcqs as Array<Record<string, unknown>>)
   296→        : Array.isArray((data as { mcqs?: unknown }).mcqs)
   297→          ? ((data as { mcqs?: Array<Record<string, unknown>> }).mcqs as Array<Record<string, unknown>>)
   298→          : [];
   299→      const answers = (data as { answers?: Record<string, string> }).answers ?? {};
   300→      for (const m of mcqs) {
   301→        const t = str(m?.topic);
   302→        if (!t) continue;
   303→        const agg = get(t);
   304→        if (!agg) continue;
   305→        const id = str(m?.id);
   306→        const correctAnswer = str(m?.correctAnswer);
   307→        const diff = str(m?.difficulty);
   308→        let correct: number | undefined;
   309→        let total: number | undefined;
   310→        let hasAcc = false;
   311→        if (id && answers[id]) {
   312→          total = 1;
   313→          correct = correctAnswer && answers[id] === correctAnswer ? 1 : 0;
   314→          hasAcc = true;
   315→        }
   316→        agg.questions.push({
   317→          source: "mcq",
   318→          itemId: item.id,
   319→          title: item.title,
   320→          difficulty: diff,
   321→          correct,
   322→          total,
   323→          hasAccuracy: hasAcc,
   324→        });
   325→      }
   326→    }
   327→  }
   328→
   329→  // 2) Flashcards
   330→  for (const set of sets) {
   331→    for (const card of set.cards) {
   332→      const t = str(card.topic);
   333→      if (!t) continue;
   334→      const agg = get(t);
   335→      if (!agg) continue;
   336→      agg.flashcards.push(card);
   337→      // FlashcardSet.topic is the set's primary topic — use as subject hint only if blank
   338→      const setTopic = str(set.topic);
   339→      if (setTopic && !agg.subject) agg.subject = setTopic;
   340→    }
   341→  }
   342→
   343→  // 3) Study sessions
   344→  for (const s of sessions) {
   345→    const t = str(s.topic);
   346→    if (!t) continue;
   347→    const agg = get(t);
   348→    if (!agg) continue;
   349→    agg.sessions.push(s);
   350→    const subj = str(s.subject);
   351→    if (subj && !agg.subject) agg.subject = subj;
   352→  }
   353→
   354→  // 4) Journal entries
   355→  for (const e of entries) {
   356→    const t = str(e.topic);
   357→    if (!t) continue;
   358→    const agg = get(t);
   359→    if (!agg) continue;
   360→    agg.journal.push(e);
   361→    const subj = str(e.subject);
   362→    if (subj && !agg.subject) agg.subject = subj;
   363→  }
   364→
   365→  // Compute scores
   366→  for (const agg of map.values()) {
   367→    // Flashcard mastery average
   368→    if (agg.flashcards.length > 0) {
   369→      const sum = agg.flashcards.reduce((a, c) => a + fcMasteryToScore(c.mastery), 0);
   370→      agg.fcScore = sum / agg.flashcards.length;
   371→    }
   372→
   373→    // Question accuracy (only entries with actual attempt data)
   374→    const scoredQs = agg.questions.filter(
   375→      (q) => q.hasAccuracy && typeof q.correct === "number" && typeof q.total === "number" && (q.total ?? 0) > 0,
   376→    );
   377→    if (scoredQs.length > 0) {
   378→      const totalC = scoredQs.reduce((a, q) => a + (q.correct ?? 0), 0);
   379→      const totalT = scoredQs.reduce((a, q) => a + (q.total ?? 0), 0);
   380→      agg.qaScore = totalT > 0 ? (totalC / totalT) * 100 : null;
   381→    } else if (agg.questions.length > 0) {
   382→      // Questions exist but no attempts recorded → neutral 50
   383→      agg.qaScore = 50;
   384→    } else {
   385→      agg.qaScore = null;
   386→    }
   387→
   388→    // Activity boost: +5/session cap 20, +3/journal cap 15 (max 35)
   389→    agg.activityBoost =
   390→      Math.min(agg.sessions.length * 5, 20) + Math.min(agg.journal.length * 3, 15);
   391→
   392→    // Weighted blend of FC (0.6) + QA (0.4) — renormalized if one missing
   393→    let blend = 0;
   394→    let weight = 0;
   395→    if (agg.fcScore !== null) {
   396→      blend += agg.fcScore * 0.6;
   397→      weight += 0.6;
   398→    }
   399→    if (agg.qaScore !== null) {
   400→      blend += agg.qaScore * 0.4;
   401→      weight += 0.4;
   402→    }
   403→    agg.blend = weight > 0 ? blend / weight : 0;
   404→
   405→    // Final = blend + activity boost, clamp 0-100
   406→    const raw = agg.blend + agg.activityBoost;
   407→    agg.score = Math.max(0, Math.min(100, Math.round(raw)));
   408→
   409→    // Classify
   410→    const hasAny =
   411→      agg.flashcards.length > 0 ||
   412→      agg.sessions.length > 0 ||
   413→      agg.journal.length > 0 ||
   414→      agg.questions.length > 0;
   415→    if (!hasAny) agg.status = "Not Started";
   416→    else if (agg.score >= 70) agg.status = "Strong";
   417→    else if (agg.score >= 40) agg.status = "Moderate";
   418→    else agg.status = "Weak";
   419→  }
   420→
   421→  return map;
   422→}
