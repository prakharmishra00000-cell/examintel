    72→function isDue(card: Flashcard, now: number): boolean {
    73→  const t = Date.parse(card.nextReview);
    74→  if (Number.isNaN(t)) return true;
    75→  return t <= now;
    76→}

    78→function relativeDay(iso: string): string {
    79→  const t = Date.parse(iso);
    80→  if (Number.isNaN(t)) return "—";
    81→  const dayMs = 24 * 60 * 60 * 1000;
    82→  const diff = Math.round((t - Date.now()) / dayMs);
    83→  if (diff <= 0) return "due now";
    84→  if (diff === 1) return "in 1 day";
    85→  return `in ${diff} days`;
    86→}

   472→interface ReviewSession {
   473→  cards: { setId: string; card: Flashcard }[];
   474→  index: number;
   475→  flipped: boolean;
   476→  reviewed: number;
   477→  masteredGained: number;

   480→
   481→function ReviewTab({ sets }: { sets: FlashcardSet[] }) {
   482→  const { updateCard } = useFlashcardStore();
   483→  const [session, setSession] = useState<ReviewSession | null>(null);
   484→  const [seed, setSeed] = useState(0); // force refresh of due check
   485→
   486→  // Snapshot due cards across all sets
   487→  const dueSnapshot = useMemo(() => {
   488→    const now = Date.now();
   489→    const out: { setId: string; card: Flashcard }[] = [];
   490→    for (const s of sets) {
   491→      for (const c of s.cards) {
   492→        if (isDue(c, now)) out.push({ setId: s.id, card: c });
   493→      }
   494→    }
   495→    return out;
   496→  }, [sets, seed]);
   497→
   498→  const startSession = useCallback(() => {
   499→    if (dueSnapshot.length === 0) return;
   500→    setSession({
   501→      cards: dueSnapshot,
   502→      index: 0,
   503→      flipped: false,
   504→      reviewed: 0,
   505→      masteredGained: 0,
   506→      finished: false,
   507→    });
   508→  }, [dueSnapshot]);
   509→
   510→  // Auto-start when cards become available (only if no session in progress)
   511→  useEffect(() => {
   512→    if (!session && dueSnapshot.length > 0) {
   513→      // don't auto-start; let the user click "Start review"
   514→    }
   515→  }, [session, dueSnapshot.length]);
   516→
   517→  const rate = useCallback(
   518→    (quality: Quality) => {
   519→      if (!session) return;
   520→      const entry = session.cards[session.index];
   521→      if (!entry) return;
   522→
   523→      const wasMastered = entry.card.mastery === "Mastered";
   524→      // Apply SM-2 update in the store (this recalculates mastery)
   525→      updateCard(entry.setId, entry.card.id, quality);
   526→      // Optimistically check whether the new state would be mastered:
   527→      // repetitions >= 3 && interval >= 7 — replicate the SM-2 logic loosely
   528→      let willMaster = wasMastered;
   529→      try {
   530→        const reps = quality < 3 ? 0 : entry.card.repetitions + 1;
   531→        const ef = Math.max(
   532→          1.3,
   533→          entry.card.easeFactor + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02))
   534→        );
   535→        let interval: number;
   536→        if (quality < 3) interval = 1;
   537→        else if (reps === 1) interval = 1;
   538→        else if (reps === 2) interval = 6;
   539→        else interval = Math.round(entry.card.interval * ef);
   540→        willMaster = reps >= 3 && interval >= 7;
   541→      } catch {
   542→        // ignore
   543→      }
   544→
   545→      const nextIndex = session.index + 1;
   546→      const isLast = nextIndex >= session.cards.length;
   547→      setSession({
   548→        ...session,
   549→        index: isLast ? session.index : nextIndex,
   550→        flipped: false,
   551→        reviewed: session.reviewed + 1,
   552→        masteredGained: session.masteredGained + (willMaster && !wasMastered ? 1 : 0),
   553→        finished: isLast,
   554→      });
   555→      setSeed((n) => n + 1);
   556→    },
   557→    [session, updateCard]
   558→  );
   559→
   560→  const next = useCallback(() => {
   561→    if (!session) return;
   562→    const ni = session.index + 1;
   563→    if (ni >= session.cards.length) {
   564→      setSession({ ...session, finished: true });
   565→    } else {
   566→      setSession({ ...session, index: ni, flipped: false });
   567→    }
   568→  }, [session]);
   569→
   570→  const prev = useCallback(() => {
   571→    if (!session) return;
   572→    setSession({ ...session, index: Math.max(0, session.index - 1), flipped: false });
   573→  }, [session]);
   574→
   575→  if (dueSnapshot.length === 0 && (!session || session.finished === false)) {
   576→    return (
   577→      <div className="space-y-6">
   578→        <StatsCard cards={sets.flatMap((s) => s.cards)} />
   579→        <EmptyState
   580→          title="No cards due right now!"
   581→          description="Spaced repetition has nothing scheduled for you. Come back later, or generate more flashcards to expand your deck."
   582→          icon={Flame}
   583→        />
   584→      </div>
   585→    );
   586→  }
   587→
   588→  if (!session) {
   589→    return (
   590→      <div className="space-y-6">
   591→        <StatsCard cards={sets.flatMap((s) => s.cards)} />
   592→        <Card className="border-violet-500/20 bg-gradient-to-br from-violet-500/10 to-fuchsia-500/5">
   593→          <CardContent className="py-8 flex flex-col items-center text-center gap-4">
   594→            <div className="h-14 w-14 rounded-full bg-gradient-to-br from-violet-500 to-fuchsia-500 flex items-center justify-center text-white shadow-lg shadow-violet-500/30">
   595→              <Brain className="h-7 w-7" />
   596→            </div>
   597→            <div>
   598→              <h3 className="text-lg font-semibold">{dueSnapshot.length} cards ready for review</h3>
   599→              <p className="text-sm text-muted-foreground max-w-md mt-1">
   600→                Each rating updates the SM-2 schedule. Rate honestly — "Again" resets the card, "Easy" spaces it out further.
   601→              </p>
   602→            </div>
   603→            <Button
   604→              onClick={startSession}
   605→              className="bg-gradient-to-r from-violet-600 to-fuchsia-600 hover:from-violet-700 hover:to-fuchsia-700 text-white border-transparent gap-2"
   606→            >
   607→              <Play className="h-4 w-4" />
   608→              Start review session
   609→            </Button>
   610→          </CardContent>
   611→        </Card>
   612→      </div>
   613→    );
   614→  }
   615→
   616→  if (session.finished) {
   617→    return (
   618→      <div className="space-y-6">
   619→        <StatsCard cards={sets.flatMap((s) => s.cards)} />
   620→        <Card className="border-emerald-500/30 bg-gradient-to-br from-emerald-500/10 via-violet-500/5 to-background">
   621→          <CardContent className="py-10 flex flex-col items-center text-center gap-3">
   622→            <motion.div
   623→              initial={{ scale: 0.5, opacity: 0 }}
   624→              animate={{ scale: 1, opacity: 1 }}
   625→              transition={{ type: "spring", stiffness: 200, damping: 15 }}
   626→              className="text-5xl"
   627→            >
   628→              <PartyPopper className="h-14 w-14 text-emerald-500 mx-auto" />
   629→            </motion.div>
   630→            <motion.div
   631→              initial={{ opacity: 0 }}
   632→              animate={{ opacity: 1 }}
   633→              transition={{ delay: 0.2 }}
   634→              className="text-3xl"
   635→            >
   636→              🎉 🌟 🎓 🎊 ✨
   637→            </motion.div>
   638→            <h3 className="text-xl font-bold mt-2">Session complete!</h3>
   639→            <p className="text-sm text-muted-foreground">
   640→              Reviewed <span className="font-semibold text-foreground">{session.reviewed}</span> cards
   641→              {" · "}
   642→              <span className="font-semibold text-emerald-500">{session.masteredGained}</span> newly mastered
   643→            </p>
   644→            <div className="flex flex-wrap items-center justify-center gap-2 mt-3">
   645→              <Button
   646→                onClick={() => setSession(null)}
   647→                variant="outline"
   648→                className="gap-1.5"
   649→              >
   650→                <RotateCcw className="h-4 w-4" />
   651→                Back to review
   652→              </Button>
   653→              <Button
   654→                onClick={startSession}
   655→                className="gap-1.5 bg-gradient-to-r from-violet-600 to-fuchsia-600 hover:from-violet-700 hover:to-fuchsia-700 text-white border-transparent"
   656→              >
   657→                <RefreshCw className="h-4 w-4" />
   658→                Review again
   659→              </Button>
   660→            </div>
   661→          </CardContent>
   662→        </Card>
   663→      </div>
   664→    );
   665→  }
   666→
   667→  const current = session.cards[session.index];
   668→  if (!current) {
   669→    // cards vanished (e.g. set deleted); end session
   670→    return (
   671→      <div className="space-y-6">
   672→        <StatsCard cards={sets.flatMap((s) => s.cards)} />
   673→        <EmptyState title="Session ended" description="No more cards available to review." icon={Check} />
   674→      </div>
   675→    );
   676→  }
   677→
   678→  const total = session.cards.length;
   679→  const progressPct = ((session.index + 1) / total) * 100;
   680→
   681→  return (
   682→    <div className="space-y-6">
   683→      <StatsCard cards={sets.flatMap((s) => s.cards)} />
   684→
   685→      <div className="space-y-3">
   686→        <div className="flex items-center justify-between gap-2">
   687→          <div className="flex items-center gap-2 text-sm text-muted-foreground">
   688→            <Flame className="h-4 w-4 text-amber-500" />
   689→            <span>Card <span className="font-semibold text-foreground">{session.index + 1}</span> of {total} due</span>
   690→          </div>
   691→          <Button
   692→            variant="ghost"
   693→            size="sm"
   694→            onClick={() => setSession(null)}
   695→            className="gap-1.5 text-muted-foreground"
   696→          >
   697→            <X className="h-3.5 w-3.5" />
   698→            End session
   699→          </Button>
   700→        </div>
   701→        <Progress value={progressPct} className="h-1.5 bg-violet-500/15" />
   702→      </div>
   703→
   704→      <AnimatePresence mode="wait">
   705→        <motion.div
   706→          key={current.card.id + (session.flipped ? "-back" : "-front")}
   707→          initial={{ opacity: 0, x: 20 }}
   708→          animate={{ opacity: 1, x: 0 }}
   709→          exit={{ opacity: 0, x: -20 }}
   710→          transition={{ duration: 0.25 }}
   711→        >
   712→          <FlashcardFace
   713→            card={current.card}
   714→            flipped={session.flipped}
   715→            onFlip={() => setSession({ ...session, flipped: !session.flipped })}
   716→          />
   717→        </motion.div>
   718→      </AnimatePresence>
   719→
   720→      <AnimatePresence>
   721→        {session.flipped && (
   722→          <motion.div
   723→            initial={{ opacity: 0, y: 10 }}
   724→            animate={{ opacity: 1, y: 0 }}
   725→            exit={{ opacity: 0, y: 10 }}
   726→            transition={{ duration: 0.2 }}
   727→          >
   728→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   729→              {RATINGS.map((r) => {
   730→                const Icon = r.icon;
   731→                return (
   732→                  <button
   733→                    key={r.label}
   734→                    onClick={() => rate(r.quality)}
   735→                    className={cn(
   736→                      "flex flex-col items-center gap-1 rounded-xl border px-3 py-3 text-sm font-medium transition shadow-sm ring-1",
   737→                      r.color,
   738→                      r.ring
   739→                    )}
   740→                  >
   741→                    <Icon className="h-4 w-4" />
   742→                    {r.label}
   743→                  </button>
   744→                );
   745→              })}
   746→            </div>
   747→            <p className="text-center text-xs text-muted-foreground mt-2">
   748→              Rate your recall — the SM-2 algorithm reschedules the card automatically.
   749→            </p>
   750→          </motion.div>
   751→        )}
   752→      </AnimatePresence>
   753→
   754→      {!session.flipped && (
   755→        <div className="flex items-center justify-between">
   756→          <Button
   757→            variant="ghost"
   758→            size="sm"
   759→            onClick={prev}
   760→            disabled={session.index === 0}
   761→            className="gap-1.5"
   762→          >
   763→            <ChevronLeft className="h-4 w-4" />
   764→            Prev
   765→          </Button>
   766→          <Button
   767→            variant="outline"
   768→            size="sm"
   769→            onClick={() => setSession({ ...session, flipped: true })}
   770→            className="gap-1.5 border-violet-500/30 text-violet-600 dark:text-violet-300"
   771→          >
   772→            Reveal answer
   773→            <ArrowRight className="h-4 w-4" />
   774→          </Button>
   775→          <Button
   776→            variant="ghost"
   777→            size="sm"
   778→            onClick={next}
   779→            disabled={session.index >= total - 1}
   780→            className="gap-1.5"
   781→          >
   782→            Skip
   783→            <ChevronRight className="h-4 w-4" />
   784→          </Button>
   785→        </div>
   786→      )}
   787→    </div>
   788→  );
   789→}
