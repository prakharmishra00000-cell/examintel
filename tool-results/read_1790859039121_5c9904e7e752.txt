   139→function todayISODate() {
   140→  // local date in YYYY-MM-DD (not UTC-shifted)
   141→  const d = new Date();
   142→  const y = d.getFullYear();
   143→  const m = String(d.getMonth() + 1).padStart(2, "0");
   144→  const day = String(d.getDate()).padStart(2, "0");
   145→  return `${y}-${m}-${day}`;
   146→}
   147→
   148→function fmtDuration(min: number) {
   149→  if (!min || min <= 0) return "0m";
   150→  const h = Math.floor(min / 60);
   151→  const m = Math.round(min % 60);
   152→  if (h && m) return `${h}h ${m}m`;
   153→  if (h) return `${h}h`;
   154→  return `${m}m`;
   155→}

   894→function moodLabelFromScore(score: number) {
   895→  if (score <= 0) return "—";
   896→  if (score >= 3.5) return "😊 Great";
   897→  if (score >= 2.5) return "🙂 Good";
   898→  if (score >= 1.5) return "😐 Okay";
   899→  return "😣 Struggle";
   900→}

   902→interface EntryGroup {
   903→  key: string;
   904→  label: string;
   905→  items: JournalEntry[];
   906→}

   908→function buildGroups(entries: JournalEntry[]): EntryGroup[] {
   909→  const sorted = [...entries].sort((a, b) => {
   910→    // most recent first by date, then createdAt
   911→    const dd = (b.date || "").localeCompare(a.date || "");
   912→    if (dd !== 0) return dd;
   913→    return (b.createdAt || "").localeCompare(a.createdAt || "");
   914→  });
   915→
   916→  const today: JournalEntry[] = [];
   917→  const yesterday: JournalEntry[] = [];
   918→  const thisWeek: JournalEntry[] = [];
   919→  const older: JournalEntry[] = [];
   920→
   921→  for (const e of sorted) {
   922→    let d: Date;
   923→    try {
   924→      d = parseISO(e.date);
   925→    } catch {
   926→      older.push(e);
   927→      continue;
   928→    }
   929→    if (isToday(d)) today.push(e);
   930→    else if (isYesterday(d)) yesterday.push(e);
   931→    else if (isThisWeek(d, { weekStartsOn: 1 })) thisWeek.push(e);
   932→    else older.push(e);
   933→  }
   934→
   935→  // sort older by month buckets? Just keep sorted list under "Older"
   936→  const groups: EntryGroup[] = [];
   937→  if (today.length) groups.push({ key: "today", label: "Today", items: today });
   938→  if (yesterday.length)
   939→    groups.push({ key: "yesterday", label: "Yesterday", items: yesterday });
   940→  if (thisWeek.length)
   941→    groups.push({ key: "thisweek", label: "Earlier this week", items: thisWeek });
   942→  if (older.length) {
   943→    // bucket older by month
   944→    const monthBuckets: Record<string, JournalEntry[]> = {};
   945→    for (const e of older) {
   946→      let d: Date;
   947→      try {
   948→        d = parseISO(e.date);
   949→      } catch {
   950→        continue;
   951→      }
   952→      const k = format(d, "MMMM yyyy");
   953→      (monthBuckets[k] ||= []).push(e);
   954→    }
   955→    // order month buckets by most recent first
   956→    const ordered = Object.entries(monthBuckets).sort((a, b) => {
   957→      const da = new Date(a[0]);
   958→      const db = new Date(b[0]);
   959→      return db.getTime() - da.getTime();
   960→    });
   961→    for (const [label, items] of ordered) {
   962→      groups.push({ key: `older-${label}`, label, items });
   963→    }
   964→  }
   965→  return groups;
   966→}
