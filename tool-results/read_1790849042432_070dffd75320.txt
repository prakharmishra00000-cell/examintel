   109→export function DependencyMapper() {
   110→  const { call } = useApi();
   111→  const setContext = useAppStore((s) => s.setContext);
   112→  const saveItem = useAppStore((s) => s.saveItem);
   113→
   114→  const [inputType, setInputType] = useState<InputType>("topic");
   115→  const [input, setInput] = useState("");
   116→  const [loading, setLoading] = useState(false);
   117→  const [report, setReport] = useState<DependencyMapReport | null>(null);
   118→  const [error, setError] = useState<string | null>(null);
   119→
   120→  async function build() {
   121→    const trimmed = input.trim();
   122→    if (!trimmed) {
   123→      toast.error("Enter something to map", {
   124→        description: "Type an exam, subject, topic, or paste a syllabus.",
   125→      });
   126→      return;
   127→    }
   128→    setLoading(true);
   129→    setError(null);
   130→    const res = await call<{ report: DependencyMapReport } | { error: string }>(
   131→      "/api/dependency/map",
   132→      { input: trimmed, inputType }
   133→    );
   134→    setLoading(false);
   135→    if (!res) return;
   136→    if ("error" in res) {
   137→      setError(res.error);
   138→      return;
   139→    }
   140→    setReport(res.report);
   141→    setContext(`Dependency map: ${res.report.root}`, "dependency");
   142→    toast.success("Dependency map ready", {
   143→      description: `${res.report.nodes.length} nodes · ${res.report.gaps.length} gaps detected`,
   144→    });
   145→  }
   146→
   147→  function quickFill(value: string) {
   148→    setInput(value);
   149→    if (/ssc|gate|upsc|cat|gre|gate|jee|neet/i.test(value)) setInputType("exam");
   150→    else if (value.length > 40) setInputType("syllabus");
   151→    else setInputType("topic");
   152→  }
   153→
   154→  function saveToResearch() {
   155→    if (!report) return;
   156→    saveItem({
   157→      type: "dependency",
   158→      title: report.root,
   159→      summary: `${report.nodes.length} nodes, ${report.gaps.length} gaps`,
   160→      data: report,
   161→    });
   162→    toast.success("Saved to My Research", {
   163→      description: `"${report.root}" — ${report.nodes.length} nodes, ${report.gaps.length} gaps`,
   164→    });
   165→  }
   166→
   167→  return (
   168→    <div className="space-y-6">
   169→      {/* Header */}
   170→      <motion.div
   171→        initial={{ opacity: 0, y: 8 }}
   172→        animate={{ opacity: 1, y: 0 }}
   173→        transition={{ duration: 0.3 }}
   174→      >
   175→        <div className="flex items-start gap-3">
   176→          <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-gradient-to-br from-violet-500/20 to-fuchsia-500/15 border border-violet-500/20">
   177→            <Network className="h-5 w-5 text-violet-500" />
   178→          </div>
   179→          <div className="flex-1">
   180→            <h1 className="text-2xl sm:text-3xl font-bold tracking-tight">
   181→              AI Exam Dependency Mapper

   183→            <p className="mt-1 text-sm text-muted-foreground">
   184→              Build a prerequisite knowledge graph. Detect missing foundations before they block you.
   185→            </p>
   186→          </div>
   187→        </div>
   188→      </motion.div>
   189→
   190→      {/* Input form */}
   191→      <Card className="border-violet-500/20 bg-gradient-to-br from-violet-500/5 via-fuchsia-500/[0.03] to-transparent">
   192→        <CardHeader>
   193→          <CardTitle className="text-base flex items-center gap-2">

   524→function NodeCard({ node, allNodes }: { node: DependencyNode; allNodes: DependencyNode[] }) {
   525→  const mastery = MASTERY_STYLE[node.mastery] ?? MASTERY_STYLE["Not Started"];
   526→  const diffKey = (node.difficulty ?? "").toLowerCase();
   527→  const diff = DIFFICULTY_STYLE[diffKey] ?? null;
   528→  const relKey = (node.examRelevance ?? "").toLowerCase();
   529→  const rel = RELEVANCE_STYLE[relKey] ?? null;
   530→
   531→  const prereqNodes = node.prerequisites
   532→    .map((id) => allNodes.find((n) => n.id === id))
   533→    .filter(Boolean) as DependencyNode[];
   534→
   535→  return (
   536→    <motion.div

   691→function groupByLevel(nodes: DependencyNode[]): { level: number; nodes: DependencyNode[] }[] {
   692→  const map = new Map<number, DependencyNode[]>();
   693→  for (const n of nodes) {
   694→    const lv = n.level ?? 0;
   695→    if (!map.has(lv)) map.set(lv, []);
   696→    map.get(lv)!.push(n);
   697→  }
   698→  return Array.from(map.entries())
   699→    .sort((a, b) => a[0] - b[0])
   700→    .map(([level, nodes]) => ({ level, nodes }));
   701→}
