   107→// ---------- Small defensive helpers ----------
   108→function str(v: unknown): string | undefined {
   109→  return typeof v === "string" && v.trim() ? v.trim() : undefined;
   110→}
   111→function numOr(v: unknown, fallback = 0): number {
   112→  return typeof v === "number" && Number.isFinite(v) ? v : fallback;
   113→}
   114→
   115→// Normalize subject strings (trim, collapse spaces, fallback "General").
   116→function normSubject(s: unknown): string {
   117→  const t = str(s);
   118→  if (!t) return "General";
   119→  return t.replace(/\s+/g, " ");
   120→}
   121→
   122→// Parse "1m 30s" / "45s" / "90" / "1.5 min" → seconds.
   123→function parseSeconds(v: unknown): number | null {
   124→  if (typeof v === "number" && Number.isFinite(v)) return v;
   125→  if (typeof v !== "string") return null;
   126→  const s = v.trim().toLowerCase();
   127→  if (!s) return null;
   128→  // pattern "Xm Ys"
   129→  let m = s.match(/(\d+)\s*m(?:\s*(\d+(?:\.\d+)?)\s*s)?/);
   130→  if (m) {
   131→    const mins = parseInt(m[1]!, 10);
   132→    const secs = m[2] ? parseFloat(m[2]) : 0;
   133→    return mins * 60 + secs;
   134→  }
   135→  // pattern "Xs"
   136→  m = s.match(/^(\d+(?:\.\d+)?)\s*s/);
   137→  if (m) return parseFloat(m[1]!);
   138→  // pattern "X min" / "X mins"
   139→  m = s.match(/^(\d+(?:\.\d+)?)\s*(?:min|mins|minutes)/);
   140→  if (m) return parseFloat(m[1]!) * 60;
   141→  // bare number
   142→  m = s.match(/^(\d+(?:\.\d+)?)/);
   143→  if (m) return parseFloat(m[1]!);
   144→  return null;
   145→}

   150→interface AttemptPoint {
   151→  id: string;
   152→  dateISO: string;
   153→  title: string;

   169→// ---------- Range helpers ----------
   170→function rangeCutoffMs(range: RangeKey): number | null {
   171→  if (range === "all") return null;
   172→  const days = range === "7" ? 7 : 30;
   173→  return Date.now() - days * 24 * 60 * 60 * 1000;
   174→}
   175→function withinRange(iso: string, cutoff: number | null): boolean {
   176→  if (cutoff === null) return true;
   177→  const t = Date.parse(iso);
   178→  if (Number.isNaN(t)) return true; // keep if unparseable (better to show than hide)
   179→  return t >= cutoff;
   180→}
   181→function rangeDays(range: RangeKey): number {
   182→  if (range === "all") return 90; // cap "all time" daily chart at last 90 days for readability
   183→  return range === "7" ? 7 : 30;
   184→}
   185→function rangeLabel(range: RangeKey): string {
   186→  return range === "all" ? "all time" : `last ${range} days`;
   187→}

   189→// ---------- Attempt extractor (papers + mcqs) ----------
   190→// Paper shape: { paper: GeneratedPaper, analysis?: PerformanceAnalysis, answers?: Record<string,string> }
   191→// MCQ shape: { set: { mcqs: GeneratedMCQ[] }, answers?: Record<string,string>, ... }
   192→function extractAttempt(item: SavedItem): AttemptPoint | null {
   193→  const data = (item.data ?? null) as Record<string, unknown> | null;
   194→  if (!data) return null;
   195→
   196→  // Use item.createdAt as the attempt date (when the user saved/attemped it).
   197→  const dateISO = item.createdAt;
   198→
   199→  // Difficulty buckets keyed by difficulty string
   200→  const diffBuckets: AttemptPoint["difficultyBreakdown"] = {
   201→    Easy: { correct: 0, incorrect: 0, unattempted: 0 },
   202→    Medium: { correct: 0, incorrect: 0, unattempted: 0 },
   203→    Hard: { correct: 0, incorrect: 0, unattempted: 0 },
   204→  };
   205→  function bucketFor(diff?: string): keyof AttemptPoint["difficultyBreakdown"] | null {
   206→    const d = (diff ?? "").trim().toLowerCase();
   207→    if (d === "easy") return "Easy";
   208→    if (d === "medium" || d === "moderate") return "Medium";
   209→    if (d === "hard" || d === "difficult") return "Hard";
   210→    return null;
   211→  }
   212→
   213→  const topicMap = new Map<
   214→    string,
   215→    { correct: number; incorrect: number; unattempted: number; total: number }
   216→  >();
   217→  function pushTopic(
   218→    topic: string | undefined,
   219→    correct: number,
   220→    incorrect: number,
   221→    unattempted: number,
   222→  ) {
   223→    const t = (topic ?? "").trim();
   224→    if (!t) return;
   225→    const cur = topicMap.get(t) ?? { correct: 0, incorrect: 0, unattempted: 0, total: 0 };
   226→    cur.correct += correct;
   227→    cur.incorrect += incorrect;
   228→    cur.unattempted += unattempted;
   229→    cur.total += correct + incorrect + unattempted;
   230→    topicMap.set(t, cur);
   231→  }
   232→
   233→  let correct = 0;
   234→  let incorrect = 0;
   235→  let unattempted = 0;
   236→  let total = 0;
   237→  let accuracy = 0;
   238→  let hasAnalysis = false;
   239→  let avgTimePerQuestionSec: number | null = null;
   240→
   241→  if (item.type === "paper") {
   242→    const paper = (data as { paper?: Record<string, unknown> }).paper;
   243→    const questions = Array.isArray(paper?.questions)
   244→      ? (paper!.questions as unknown as Array<Record<string, unknown>>)
   245→      : Array.isArray((data as { questions?: unknown }).questions)
   246→        ? ((data as { questions?: Array<Record<string, unknown>> }).questions as Array<Record<string, unknown>>)
   247→        : [];
   248→    const analysis = (data as { analysis?: PerformanceAnalysis }).analysis;
   249→    const answers =
   250→      (data as { answers?: Record<string, string> }).answers ?? undefined;
   251→
   252→    if (analysis) {
   253→      hasAnalysis = true;
   254→      correct = numOr(analysis.correct, 0);
   255→      incorrect = numOr(analysis.incorrect, 0);
   256→      unattempted = numOr(analysis.unattempted, 0);
   257→      total = correct + incorrect + unattempted;
   258→      const denom = total > 0 ? total : numOr(analysis.maxMarks, 0);
   259→      accuracy = total > 0
   260→        ? (correct / total) * 100
   261→        : typeof analysis.accuracy === "number"
   262→          ? analysis.accuracy
   263→          : 0;
   264→      // difficulty-wise from analysis if present (correct/total only)
   265→      const dw = analysis.difficultyWise ?? [];
   266→      for (const d of dw) {
   267→        const b = bucketFor(d.difficulty);
   268→        if (!b) continue;
   269→        const t = numOr(d.total, 0);
   270→        const c = numOr(d.correct, 0);
   271→        // If we have per-question answers below, prefer that; otherwise split here.
   272→        diffBuckets[b].correct += c;
   273→        diffBuckets[b].incorrect += Math.max(0, t - c);
   274→        diffBuckets[b].unattempted += 0;
   275→      }
   276→      // topic-wise from analysis (correct/total only — split remaining as incorrect)
   277→      for (const tw of analysis.topicWise ?? []) {
   278→        const t = numOr(tw.total, 0);
   279→        const c = numOr(tw.correct, 0);
   280→        pushTopic(tw.topic, c, Math.max(0, t - c), 0);
   281→      }
   282→      avgTimePerQuestionSec = parseSeconds(analysis.avgTimePerQuestion);
   283→    }
   284→
   285→    // If we have per-question answers, recompute granular difficulty/topic breakdown
   286→    if (questions.length > 0 && answers && typeof answers === "object") {
   287→      // Reset difficulty counters if analysis-derived (so we don't double-count).
   288→      if (hasAnalysis) {
   289→        diffBuckets.Easy = { correct: 0, incorrect: 0, unattempted: 0 };
   290→        diffBuckets.Medium = { correct: 0, incorrect: 0, unattempted: 0 };
   291→        diffBuckets.Hard = { correct: 0, incorrect: 0, unattempted: 0 };
   292→        topicMap.clear();
   293→      }
   294→      let gc = 0,
   295→        gi = 0,
   296→        gu = 0;
   297→      for (const q of questions) {
   298→        const qid = str(q.id);
   299→        const diff = str(q.difficulty);
   300→        const topic = str(q.topic);
   301→        const correctAnswer = str(q.correctAnswer);
   302→        const picked = qid ? answers[qid] : undefined;
   303→        const b = bucketFor(diff);
   304→        const hasPicked = typeof picked === "string" && picked.trim().length > 0;
   305→        if (hasPicked && correctAnswer) {
   306→          if (picked === correctAnswer) {
   307→            gc++;
   308→            if (b) diffBuckets[b].correct++;
   309→            pushTopic(topic, 1, 0, 0);
   310→          } else {
   311→            gi++;
   312→            if (b) diffBuckets[b].incorrect++;
   313→            pushTopic(topic, 0, 1, 0);
   314→          }
   315→        } else {
   316→          gu++;
   317→          if (b) diffBuckets[b].unattempted++;
   318→          pushTopic(topic, 0, 0, 1);
   319→        }
   320→      }
   321→      // Prefer granular counts if any exist
   322→      if (gc + gi + gu > 0) {
   323→        correct = gc;
   324→        incorrect = gi;
   325→        unattempted = gu;
   326→        total = gc + gi + gu;
   327→        accuracy = total > 0 ? (gc / total) * 100 : 0;
   328→        hasAnalysis = true;
   329→      }
   330→    }
   331→  } else if (item.type === "mcq") {
   332→    const set = (data as { set?: Record<string, unknown> }).set;
   333→    const mcqs = Array.isArray(set?.mcqs)
   334→      ? (set!.mcqs as unknown as Array<Record<string, unknown>>)
   335→      : Array.isArray((data as { mcqs?: unknown }).mcqs)
   336→        ? ((data as { mcqs?: Array<Record<string, unknown>> }).mcqs as Array<Record<string, unknown>>)
   337→        : [];
   338→    const answers =
   339→      (data as { answers?: Record<string, string> }).answers ?? undefined;
   340→
   341→    if (mcqs.length === 0) return null;
   342→
   343→    // Reset (no analysis-derived counts for mcqs)
   344→    let gc = 0,
   345→      gi = 0,
   346→      gu = 0;
   347→    for (const q of mcqs) {
   348→      const qid = str(q.id);
   349→      const diff = str(q.difficulty);
   350→      const topic = str(q.topic) ?? str(set?.topic);
   351→      const correctAnswer = str(q.correctAnswer);
   352→      const picked = qid ? answers?.[qid] : undefined;
   353→      const b = bucketFor(diff);
   354→      const hasPicked = typeof picked === "string" && picked.trim().length > 0;
   355→      if (hasPicked && correctAnswer) {
   356→        if (picked === correctAnswer) {
   357→          gc++;
   358→          if (b) diffBuckets[b].correct++;
   359→          pushTopic(topic, 1, 0, 0);
   360→        } else {
   361→          gi++;
   362→          if (b) diffBuckets[b].incorrect++;
   363→          pushTopic(topic, 0, 1, 0);
   364→        }
   365→      } else {
   366→        gu++;
   367→        if (b) diffBuckets[b].unattempted++;
   368→        pushTopic(topic, 0, 0, 1);
   369→      }
   370→    }
   371→    correct = gc;
   372→    incorrect = gi;
   373→    unattempted = gu;
   374→    total = mcqs.length;
   375→    accuracy = total > 0 ? (gc / total) * 100 : 0;
   376→    hasAnalysis = true;
   377→    avgTimePerQuestionSec = null; // mcq sets don't carry avgTimePerQuestion
   378→  } else {
   379→    return null;
   380→  }
   381→
   382→  if (total === 0 && !hasAnalysis) return null;
   383→
   384→  return {
   385→    id: item.id,
   386→    dateISO,
   387→    title: item.title,
   388→    type: item.type,
   389→    accuracy: Math.round(accuracy * 10) / 10,
   390→    correct,
   391→    incorrect,
   392→    unattempted,
   393→    total,
   394→    hasAnalysis,
   395→    difficultyBreakdown: diffBuckets,
   396→    topicMap,
   397→    avgTimePerQuestionSec,
   398→  };
   399→}

   415→// ---------- Custom tooltip ----------
   416→interface TipPayloadItem {
   417→  name?: string;
   418→  value?: number | string;
   419→  color?: string;
   420→  dataKey?: string | number;
   421→  payload?: Record<string, unknown>;
   422→}
   423→function ChartTooltip({
   424→  active,
   425→  payload,
   426→  label,
   427→  suffix = "",
   428→}: {
   429→  active?: boolean;
   430→  payload?: TipPayloadItem[];
   431→  label?: string | number;
   432→  suffix?: string;
   433→}) {
   434→  if (!active || !payload || payload.length === 0) return null;
   435→  return (
   436→    <div className="rounded-lg border border-border bg-popover/95 backdrop-blur px-3 py-2 shadow-lg text-xs">
   437→      {label !== undefined && (
   438→        <div className="font-medium text-foreground mb-1">{String(label)}</div>
   439→      )}
   440→      <div className="space-y-1">
   441→        {payload.map((p, i) => (
   442→          <div key={i} className="flex items-center gap-2">
   443→            <span
   444→              className="h-2 w-2 rounded-full inline-block"
   445→              style={{ backgroundColor: p.color ?? CHART.violet }}
   446→            />
   447→            <span className="text-muted-foreground">{p.name ?? p.dataKey}</span>
   448→            <span className="ml-auto font-semibold tabular-nums text-foreground">
   449→              {typeof p.value === "number" ? p.value.toLocaleString() : p.value}
   450→              {suffix}
   451→            </span>
   452→          </div>
   453→        ))}
   454→      </div>
   455→    </div>
   456→  );
   457→}

   548→      dailyMap.set(key, {
   549→        date: d.toLocaleDateString(undefined, { month: "short", day: "numeric" }),
   550→        sessions: 0,
   551→        journal: 0,
   552→      });
   553→    }
   554→    const normDay = (iso: string) => {
   555→      const d = new Date(iso);

   557→      return d.toISOString().slice(0, 10);
   558→    };
