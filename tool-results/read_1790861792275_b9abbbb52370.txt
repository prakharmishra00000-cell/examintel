  1213→// Formula Sheet lookup — searches the built-in SEED_FORMULAS by
  1214→// keyword (name / topic / subject match). Returns 3–5 matches, or

  1216→// ============================================================
  1217→function mockFormulasLookup(query: string): unknown {
  1218→  const lower = query.toLowerCase();
  1219→
  1220→  // Tokenize the query into candidate keywords (drop short / stop words).
  1221→  const stop = new Set([
  1222→    "the", "a", "an", "of", "for", "and", "or", "to", "in", "is", "are",
  1223→    "what", "whats", "formula", "formulas", "formulae", "lookup", "search",
  1224→    "find", "me", "please", "give", "show", "list", "all", "with",
  1225→  ]);
  1226→  const tokens = lower
  1227→    .split(/[^a-z0-9&+-]+/i)
  1228→    .map((t) => t.trim())
  1229→    .filter((t) => t.length > 1 && !stop.has(t));
  1230→
  1231→  const matched = SEED_FORMULAS.filter((f) => {
  1232→    const hay = `${f.subject} ${f.topic} ${f.name} ${f.description}`.toLowerCase();
  1233→    return tokens.some((tok) => hay.includes(tok));
  1234→  });
  1235→
  1236→  // De-dup by id (in case the same formula matches multiple tokens).
  1237→  const seen = new Set<string>();
  1238→  const unique = matched.filter((f) =>
  1239→    seen.has(f.id) ? false : (seen.add(f.id), true)
  1240→  );
  1241→
  1242→  // 3–5 matches when possible; if fewer keywords match, fall back to all.
  1243→  let result = unique;
  1244→  if (unique.length < 3) {
  1245→    result = SEED_FORMULAS.slice();
  1246→  }
  1247→  const capped = result.slice(0, 5);

  1249→  return {
